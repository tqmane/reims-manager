cmake_minimum_required(VERSION 3.24)
project(ReimsWifiMacNat LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(packet-tests packet-tests.cpp)
target_include_directories(packet-tests PRIVATE "${GSL_SOURCE}/include")
add_executable(reims-wifi-macnat main.cpp "${NDISAPI_SOURCE}/ndisapi/ndisapi.cpp")
target_compile_definitions(reims-wifi-macnat PRIVATE _LIB _WINDOWS WIN32 WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(reims-wifi-macnat PRIVATE "${NDISAPI_SOURCE}/include" "${GSL_SOURCE}/include")
target_link_libraries(reims-wifi-macnat PRIVATE ws2_32 iphlpapi advapi32 wintrust crypt32 ole32)
add_executable(ipv6-tests ipv6-tests.cpp)
target_include_directories(ipv6-tests PRIVATE "${GSL_SOURCE}/include")
enable_testing()
add_test(NAME packets COMMAND packet-tests)
add_test(NAME ipv6 COMMAND ipv6-tests)
